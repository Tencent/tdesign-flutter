Widget _buildSearchBarWithShape(BuildContext context) {
    return Column(
      children: [
        TSearchBar(
          hintText: '搜索预设文案',
          onChanged: (String text) {
            setState(() {
              inputText = text;
            });
          },
        ),
        const SizedBox(height: 16),
        Theme(
          data: Theme.of(context).mergeExtension(
            const TSearchBarThemeData(variant: TSearchBarVariant.round),
          ),
          child: TSearchBar(
            hintText: '搜索预设文案',
            onChanged: (String text) {
              setState(() {
                inputText = text;
              });
            },
          ),
        ),
      ],
    );
  }
