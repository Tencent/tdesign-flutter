
  Widget _loadingWithoutTextToast(BuildContext context) {
    return TButton(
      onTap: () {
        TToast.showLoadingWithoutText(context: context);
      },
      size: TButtonSize.large,
      type: TButtonType.outline,
      theme: TButtonTheme.primary,
      isBlock: true,
      text: '加载状态（无文案）',
    );
  }
