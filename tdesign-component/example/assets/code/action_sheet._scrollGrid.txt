Widget _scrollGrid(BuildContext context) => _trigger(
    label: '多行滚动宫格型',
    onPressed: () => TActionSheet.showGrid(
      context,
      cancelText: 'Cancel',
      items: _scrollGridItems(),
      layout: const TActionSheetGridLayout.scroll(
        count: 10,
        rows: 2,
        itemMinWidth: 80,
      ),
      onSelected: (item) => _showSelection(context, item),
    ),
  );
