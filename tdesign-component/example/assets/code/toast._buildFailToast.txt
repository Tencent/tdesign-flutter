Widget _buildFailToast(BuildContext context) {
    return SizedBox(
      width: double.infinity,
      child: TButton(
        child: const Text('失败'),
        size: TButtonSize.large,
        variant: TButtonVariant.outline,
        colorScheme: TButtonColorScheme.danger,
        onPressed: () {
          TToast.showFail('操作失败', context: context);
        },
      ),
    );
  }
