
  Widget buildPopup(BuildContext context) {

      title: '弹窗模式',
      note: '点击弹出，确认后关闭',
      arrow: true,
      onClick: (_) => _showInPopup(

        child: Column(mainAxisSize: MainAxisSize.min, children: [

        ]),
      ),
    );
