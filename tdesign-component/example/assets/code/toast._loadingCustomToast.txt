
  Widget _loadingCustomToast(BuildContext context) {
    return TButton(
      onTap: () {
        TToast.showLoading(
          context: context,
          customWidget: Container(
            width: 50,
            height: 20,
            child: const TText('自定义加载'),
            color: TTheme.of(context).brandColor1,
          ),
        );
      },
      size: TButtonSize.large,
      type: TButtonType.outline,
      theme: TButtonTheme.primary,
      isBlock: true,
      text: '加载状态',
    );
  }
