
Widget _buildPadding(BuildContext context) {
  var style = TCellStyle(context: context);
  style.padding = const EdgeInsets.all(30);
  return TCellGroup(
    theme: TCellGroupTheme.cardTheme,
    cells: [
      TCell(
        arrow: true,
        title: 'padding-all-30',
        style: style,
        onClick: (cell) {
          print('padding-all-30');
        },
      ),
    ],
  );
}
