import 'package:flutter/material.dart';
import 'package:tdesign_flutter/tdesign_flutter.dart';

class SliderNonZeroExample extends StatefulWidget {
  const SliderNonZeroExample({super.key});

  @override
  State<SliderNonZeroExample> createState() => _SliderNonZeroExampleState();
}

class _SliderNonZeroExampleState extends State<SliderNonZeroExample> {
  Widget _buildNonZero(BuildContext context) => _panel(
    TSlider(
      key: const ValueKey('slider-non-zero'),
      value: _nonZero,
      min: 20,
      max: 100,
      divisions: 4,
      showScaleValue: true,
      scaleFormatter: _integer,
      onChanged: (value) => setState(() => _nonZero = value),
    ),
  );

  Widget _panel(Widget child) =>
      Padding(padding: const EdgeInsets.symmetric(horizontal: 8), child: child);

  double _nonZero = 30;

  static String _integer(double value) => value.round().toString();

  @override
  Widget build(BuildContext context) {
    return _buildNonZero(context);
  }
}
