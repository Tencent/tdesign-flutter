
  Widget buildMultiColumn(BuildContext context) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text('选中: ${selectedSeason.isEmpty ? "未选择" : selectedSeason}',
            style: TextStyle(fontSize: 14, color: TTheme.of(context).textColorSecondary)),
        SizedBox(height: 8),
        _pickerCard(
          context,
          child: TPicker(items: seasonData, initialValue: ['spring', 2],
              onChange: (v) => setState(() =>
                  selectedSeason = '${v.labels.first} ${v.labels.last}')),
        ),
      ],
    );
  }
