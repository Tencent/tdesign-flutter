
  Widget _buildPopFromRight(BuildContext context) {
    return TButton(
      text: 'right',
      isBlock: true,
      theme: TButtonTheme.primary,
      type: TButtonType.outline,
      size: TButtonSize.large,
      onTap: () {
        TPopup.show(
          context,
          options: TPopupOptions.right(
              width: 280,
              child: Container(
                color: TTheme.of(context).bgColorContainer,
              )),
        );
      },
    );
  }
