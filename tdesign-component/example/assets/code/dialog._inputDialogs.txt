Widget _inputDialogs(BuildContext context) {
    return _scenarios([
      _trigger('输入类-无描述', () {
        TDialog.show<bool>(
          context,
          dialog: TDialog(
            title: const Text('带输入框对话框'),
            content: _dialogInput(context, topPadding: 8),
            actions: _actions(),
          ),
        );
      }),
      _trigger('输入类-带描述', () {
        TDialog.show<bool>(
          context,
          dialog: TDialog(
            title: const Text('带输入框对话框'),
            content: Column(
              children: [
                const Text(_description),
                const SizedBox(height: 16),
                _dialogInput(context),
              ],
            ),
            actions: _actions(),
          ),
        );
      }),
    ]);
  }
