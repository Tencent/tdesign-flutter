Widget _errorNoticeBar(BuildContext context) {
  return Theme(
    data: Theme.of(context).mergeExtension(
      const TNoticeBarThemeData(
        variant: TNoticeBarVariant.error,
        prefixIcon: TIcons.error_circle_filled,
      ),
    ),
    child: const TNoticeBar(
      content: '这是一条错误的通知信息',
    ),
  );
}
