import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:tdesign_flutter/tdesign_flutter.dart' hide TIcons;
import 'package:tdesign_flutter_icons/tdesign_flutter_icons.dart';
import 'package:url_launcher/link.dart';

class IconThemeExample extends StatelessWidget {
  const IconThemeExample({super.key});

  Widget _buildIconThemeExample(BuildContext context) {
    // 通过 Flutter IconTheme 统一控制子树图标默认尺寸和颜色。
    return IconTheme(
      data: IconThemeData(size: 36, color: context.tTheme.brandColor),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        crossAxisAlignment: CrossAxisAlignment.center,
        children: [
          const Row(
            mainAxisSize: MainAxisSize.min,
            children: [
              TIcon(TIcons.home_filled),
              SizedBox(width: 16),
              TIcon(TIcons.setting),
              SizedBox(width: 16),
              TIcon(TIcons.notification),
            ],
          ),
          const SizedBox(height: 4),
          Text(
            '↑ 继承 IconTheme 默认 size=36 和品牌色',
            style: TextStyle(
              fontSize: 12,
              color: context.tTheme.textColorSecondary,
            ),
          ),
        ],
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return _buildIconThemeExample(context);
  }
}
