Widget _buildThemed(BuildContext context) {
    return Theme(
      data: Theme.of(
        context,
      ).mergeExtension(const TPickerThemeData(height: 160, itemCount: 3)),
      child: TPicker(
        items: _columns,
        value: _columnValue,
        onChanged: (value) => setState(() => _columnValue = value.values),
      ),
    );
  }
