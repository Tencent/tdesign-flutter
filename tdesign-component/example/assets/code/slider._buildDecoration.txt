        data: Theme.of(context).mergeExtension(
          TSliderThemeData(
            decoration: BoxDecoration(
              color: context.tTheme.bgColorSecondaryContainer,
              borderRadius: BorderRadius.circular(context.tTheme.radiusDefault),
            ),
          ),
        ),

          value: range,
          min: 0,
