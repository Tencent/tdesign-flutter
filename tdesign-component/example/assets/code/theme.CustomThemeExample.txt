import 'package:flutter/material.dart';
import 'package:tdesign_flutter/tdesign_flutter.dart';

class CustomThemeExample extends StatefulWidget {
  const CustomThemeExample({super.key});

  @override
  State<CustomThemeExample> createState() => _CustomThemeExampleState();
}

class _CustomThemeExampleState extends State<CustomThemeExample> {
  static const _colors = {'默认色': '#0052D9', '绿色': '#2BA471', '自定义色': '#834EC2'};
  String _brand = '默认色';
  double _radius = 6;
  double _iconSpacing = 4;
  String _message = '';

  @override
  Widget build(BuildContext context) {
    final token = TThemeData.fromJson(
      'brand',
      '{"brand":{"color":{"brandColor":"${_colors[_brand]}"},"radius":{"radiusDefault":$_radius}}}',
    )!;
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: TThemeBuilder.light(
        token,
      ).mergeExtension(TButtonThemeData(iconTextSpacing: _iconSpacing)),
      darkTheme: TThemeBuilder.dark(
        token,
      ).mergeExtension(TButtonThemeData(iconTextSpacing: _iconSpacing)),
      home: Builder(
        builder: (context) {
          final theme = Theme.of(context);
          return Scaffold(
            appBar: const TNavBar(title: Text('自定义主题')),
            body: ListView(
              padding: const EdgeInsets.all(16),
              children: [
                const TText('应用品牌色'),
                const SizedBox(height: 8),
                Wrap(
                  spacing: 8,
                  runSpacing: 8,
                  children: [
                    for (final name in _colors.keys)
                      TButton(
                        onPressed: () => setState(() => _brand = name),
                        variant: _brand == name
                            ? TButtonVariant.fill
                            : TButtonVariant.outline,
                        colorPreset: TButtonColorPreset.primary,
                        child: Text(name),
                      ),
                  ],
                ),
                const SizedBox(height: 16),
                Wrap(
                  spacing: 8,
                  runSpacing: 8,
                  children: [
                    TButton(
                      onPressed: () =>
                          setState(() => _radius = _radius == 6 ? 16 : 6),
                      child: Text('全局圆角：${_radius.toInt()}'),
                    ),
                    TButton(
                      onPressed: () => setState(
                        () => _iconSpacing = _iconSpacing == 4 ? 16 : 4,
                      ),
                      child: Text('图标间距：${_iconSpacing.toInt()}'),
                    ),
                  ],
                ),
                const SizedBox(height: 24),
                TText('全局样式：圆角 ${_radius.toInt()}'),
                const SizedBox(height: 8),
                Align(
                  alignment: Alignment.centerLeft,
                  child: TButton(
                    onPressed: () => setState(() => _message = '全局按钮'),
                    colorPreset: TButtonColorPreset.primary,
                    icon: const Icon(TIcons.add),
                    child: const Text('全局按钮'),
                  ),
                ),
                const SizedBox(height: 16),
                const TText('局部覆盖：圆角固定 24，保留品牌色'),
                const SizedBox(height: 8),
                Theme(
                  data: theme.mergeExtension(
                    context.tTheme.copyWith(radiusMap: {'radiusDefault': 24}),
                  ),
                  child: Align(
                    alignment: Alignment.centerLeft,
                    child: TButton(
                      onPressed: () => setState(() => _message = '局部按钮'),
                      colorPreset: TButtonColorPreset.primary,
                      icon: const Icon(TIcons.add),
                      child: const Text('局部按钮'),
                    ),
                  ),
                ),
                const SizedBox(height: 16),
                const TText('实例参数：仅改变尺寸'),
                const SizedBox(height: 8),
                Align(
                  alignment: Alignment.centerLeft,
                  child: TButton(
                    onPressed: () => setState(() => _message = '大号按钮'),
                    size: TButtonSize.large,
                    colorPreset: TButtonColorPreset.primary,
                    icon: const Icon(TIcons.add),
                    child: const Text('大号按钮'),
                  ),
                ),
                if (_message.isNotEmpty) ...[
                  const SizedBox(height: 16),
                  TText('当前点击：$_message'),
                ],
              ],
            ),
          );
        },
      ),
    );
  }
}
