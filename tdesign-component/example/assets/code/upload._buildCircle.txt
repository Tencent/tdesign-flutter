Widget _buildCircle(BuildContext context) {
    return Theme(
      data: Theme.of(context).mergeExtension(
        const TUploadThemeData(
          variant: TUploadVariant.circle,
          itemSize: 72,
          spacing: 12,
        ),
      ),
      child: _section(
        TUpload(
          files: _files,
          maxFiles: 4,
          onChanged: (files) => setState(() => _files = files),
        ),
      ),
    );
  }
