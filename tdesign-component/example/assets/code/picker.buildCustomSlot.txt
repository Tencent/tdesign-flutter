
  Widget buildCustomSlot(BuildContext context) {
    final theme = TTheme.of(context);
    return TButton(
      text: '打开自定义头部弹窗',
      isBlock: true,
      type: TButtonType.outline,
      onTap: () {
        TPickerValue? draft;
        const initial = ['GD', 'SZ', 'NS'];
        TPopup.show(
          context,
          options: TPopupOptions.bottom(
            titleWidget: Row(
              mainAxisSize: MainAxisSize.min,
              children: [
                Icon(TIcons.location, size: 18, color: theme.brandNormalColor),
                const SizedBox(width: 4),
                Text(
                  '选择地区',
                  style: TextStyle(
                    fontSize: 16,
                    fontWeight: FontWeight.w600,
                    color: theme.fontGyColor1,
                  ),
                ),
              ],
            ),
            cancelBuilder: (_, close) => TToolbarPressable(
              onTap: close,
              child: Icon(TIcons.close, size: 22, color: theme.fontGyColor2),
            ),
            confirmBuilder: (_, close) => TToolbarPressable(
              onTap: close,
              child: Icon(TIcons.check, size: 22, color: theme.brandNormalColor),
            ),
            child: Material(
              color: theme.bgColorContainer,
              child: SafeArea(
                top: false,
                child: TPicker(
                  items: linkedItems,
                  initialValue: initial,
                  onChange: (value) => draft = value,
                ),
              ),
            ),
          ),
        );
      },
    );
  }
