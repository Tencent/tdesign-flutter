Widget _buildConfirmDialog(BuildContext context) {
    return _trigger(
      text: '单操作确认弹窗',
      onPressed: () {
        TDialog.show<bool>(
          context,
          dialog: const TConfirmDialog(
            title: '弹窗标题',
            content: '告知当前状态、信息和解决方法',
          ),
        );
      },
    );
  }
