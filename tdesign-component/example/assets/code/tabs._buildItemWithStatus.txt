
  Widget _buildItemWithStatus(BuildContext context) {
    var tabs = [
      const TTab(text: '选中'),
      const TTab(text: '默认'),
      const TTab(text: '禁用', enable: false),
    ];
    return TTabBar(
      tabs: tabs,
      controller: TabController(length: tabs.length, vsync: this),
      showIndicator: true,
    );
  }
