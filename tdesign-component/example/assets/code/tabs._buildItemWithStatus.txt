Widget _buildItemWithStatus(BuildContext context) {
    var tabs = [
      const TTab(text: '选中'),
      const TTab(text: '默认'),
      const TTab(text: '禁用', enabled: false),
    ];
    return TTabsBar(
      tabs: tabs,
      controller: _demoController('status', tabs.length),
      indicator: _indicator(context),
    );
  }
