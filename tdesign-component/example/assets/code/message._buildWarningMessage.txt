
  Widget _buildWarningMessage(BuildContext context) {
    return TButton(
        isBlock: true,
        text: '警示通知',
        size: TButtonSize.large,
        type: TButtonType.outline,
        theme: TButtonTheme.primary,
        onTap: () {
          TMessage.showMessage(
            context: context,
            visible: true,
            icon: true,
            content: _commonContent,
            theme: MessageTheme.warning,
            duration: 3000,
          );
        });
  }
