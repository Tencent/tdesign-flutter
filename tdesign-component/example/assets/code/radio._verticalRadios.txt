Widget _verticalRadios(BuildContext context) {
    return TRadioGroup<int>(
      value: _verticalValue,
      options: _options,
      onChanged: (value) {
        setState(() => _verticalValue = _verticalValue == value ? null : value);
      },
      showDivider: true,
      titleMaxLines: 3,
      subTitleMaxLines: 5,
    );
  }
