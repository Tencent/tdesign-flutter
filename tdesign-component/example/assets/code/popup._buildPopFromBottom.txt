
  Widget _buildPopFromBottom(BuildContext context) {
    return TButton(
      text: 'bottom',
      isBlock: true,
      theme: TButtonTheme.primary,
      type: TButtonType.outline,
      size: TButtonSize.large,
      onTap: () {
        TPopup.show(
          context,
          options: TPopupOptions.bottom(
              height: 240,
              headerBuilder: null,
              child: Container(
                color: TTheme.of(context).bgColorContainer,
                height: 240,
              )),
        );
      },
    );
  }
