Widget _buildTitle(BuildContext context) => Column(
    children: [
      TCellGroup(
        cells: [
          _cell(
            this.context,
            'title',
            '带标题选择器',
            _cities,
            value: _values['title']!,
            onConfirm: (value) => setState(() => _values['title'] = value),
            popupTitle: '选择地区',
          ),
        ],
      ),
      SizedBox(height: context.tTheme.spacer16),
      TCellGroup(
        cells: [
          _cell(
            this.context,
            'without-title',
            '无标题选择器',
            _cities,
            value: _values['without-title']!,
            onConfirm: (value) =>
                setState(() => _values['without-title'] = value),
          ),
        ],
      ),
    ],
  );
