Widget _buildCenterSearchBar(BuildContext context) {
    return Theme(
      data: Theme.of(context).mergeExtension(
        const TSearchBarThemeData(textAlignment: TSearchBarAlignment.center),
      ),
      child: TSearchBar(
        hintText: '搜索预设文案',
        onChanged: (String text) {
          setState(() {
            inputText = text;
          });
        },
      ),
    );
  }
