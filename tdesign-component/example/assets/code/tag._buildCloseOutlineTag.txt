Widget _buildCloseOutlineTag(BuildContext context) {
    return Wrap(
      spacing: 8,
      children: _closableOutlineTags
          .map((text) => TTag(
                text,
                variant: TTagVariant.outline,
                needCloseIcon: true,
                onCloseTap: () =>
                    setState(() => _closableOutlineTags.remove(text)),
              ))
          .toList(),
    );
  }
