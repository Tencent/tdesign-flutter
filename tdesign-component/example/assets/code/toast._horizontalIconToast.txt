
  Widget _horizontalIconToast(BuildContext context) {
    return TButton(
      onTap: () {
        TToast.showIconText(
          '带横向图标',
          icon: TIcons.check_circle,
          context: context,
        );
      },
      size: TButtonSize.large,
      type: TButtonType.outline,
      theme: TButtonTheme.primary,
      isBlock: true,
      text: '带横向图标',
    );
  }
