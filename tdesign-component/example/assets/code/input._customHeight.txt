
  Widget _customHeight(BuildContext context) {
    var controller = TextEditingController();
    return Container(
      color: Colors.yellow,
      alignment: Alignment.center,
      height: 90,
      child: SizedBox(
        height: 60,
        child: TInput(
          size: TInputSize.small,
          leftLabel: '标签文字',
          controller: controller,
          hintText: '请输入文字',
          needClear: true,
        ),
      ),
    );
  }
