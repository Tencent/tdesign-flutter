
  Widget _buildItemWithOutlineNormal(BuildContext context) {
    var tabs = [
      const TTab(text: '选项1'),
      const TTab(text: '选项2'),
      const TTab(text: '选项3'),
      const TTab(text: '选项4'),
    ];
    return TTabBar(
      tabs: tabs,
      outlineType: TTabBarOutlineType.capsule,
      controller: TabController(length: tabs.length, vsync: this),
      showIndicator: false,
    );
  }
