Widget _sizes(BuildContext context) {
    return Column(
      children: [
        for (final size in TCheckboxSize.values)
          TCheckbox(
            value: true,
            size: size,
            title: switch (size) {
              TCheckboxSize.small => '小尺寸',
              TCheckboxSize.medium => '中尺寸',
              TCheckboxSize.large => '大尺寸',
            },
            onChanged: (_) {},
          ),
      ],
    );
  }
