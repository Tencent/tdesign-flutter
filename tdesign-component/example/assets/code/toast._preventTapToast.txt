
  Widget _preventTapToast(BuildContext context) {
    return TButton(
      onTap: () {
        TToast.showText(
          '轻提示文字内容',
          context: context,
          preventTap: true,
          backgroundColor: Colors.black.withOpacity(0.7),
        );
      },
      size: TButtonSize.large,
      type: TButtonType.outline,
      theme: TButtonTheme.primary,
      isBlock: true,
      text: '禁止滚动+点击',
    );
  }
