
  Widget _buildItemWithContent(BuildContext context) {
    var tabController = TabController(length: 3, vsync: this);
    return SizedBox(
      height: 120 + 48,
      child: Column(
        children: [
          TTabBar(
            tabs: subList(3),
            controller: tabController,
            showIndicator: true,
            isScrollable: false,
          ),
          Container(
            height: 120,
            color: TTheme.of(context).bgColorContainer,
            child: TTabBarView(
              children: _getTabViews(),
              controller: tabController,
            ),
          )
        ],
      ),
    );
  }
