
  Widget _buildSuccessMessage(BuildContext context) {
    return TButton(
      isBlock: true,
      text: '成功通知',
      size: TButtonSize.large,
      type: TButtonType.outline,
      theme: TButtonTheme.primary,
      onTap: () {
        TMessage.showMessage(
          context: context,
          visible: true,
          icon: true,
          content: _commonContent,
          theme: MessageTheme.success,
          duration: 3000,
        );
      },
    );
  }
