Widget _buildCloseDialog(BuildContext context) {
    return _trigger(
      text: '带关闭按钮',
      onPressed: () {
        TDialog.show<void>(
          context,
          dialog: const TDialog(
            title: Text('弹窗标题'),
            content: Text('可通过右上角按钮关闭。'),
            showCloseButton: true,
          ),
        );
      },
    );
  }
