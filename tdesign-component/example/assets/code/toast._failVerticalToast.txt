
  Widget _failVerticalToast(BuildContext context) {
    return TButton(
      onTap: () {
        TToast.showFail(
          '失败文案',
          direction: IconTextDirection.vertical,
          context: context,
        );
      },
      size: TButtonSize.large,
      type: TButtonType.outline,
      theme: TButtonTheme.primary,
      isBlock: true,
      text: '失败提示(竖向)',
    );
  }
