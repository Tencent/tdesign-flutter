
  Widget _buildSingleHandleWithScale(BuildContext context) {

      sliderThemeData: TSliderThemeData(
        context: context,
        showScaleValue: true,
        divisions: 5,
        min: 0,
        max: 100,

      ),
      value: 60,
      onChanged: (value) {},
    );
  }
