
  Widget _multipleToast(BuildContext context) {
    return TButton(
      onTap: () {
        TToast.showText('最多一行展示十个汉字宽度限制最多不超过三行文字', context: context);
      },
      size: TButtonSize.large,
      type: TButtonType.outline,
      theme: TButtonTheme.primary,
      isBlock: true,
      text: '多行文字',
    );
  }
