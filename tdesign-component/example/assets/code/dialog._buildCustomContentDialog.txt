Widget _buildCustomContentDialog(BuildContext context) {
    return _trigger(
      text: '自定义内容',
      onPressed: () {
        TDialog.show<void>(
          context,
          dialog: const TDialog(
            title: Text('选择通知方式'),
            content: Column(
              children: [
                ListTile(
                  contentPadding: EdgeInsets.zero,
                  leading: Icon(Icons.email_outlined),
                  title: Text('邮件通知'),
                ),
                ListTile(
                  contentPadding: EdgeInsets.zero,
                  leading: Icon(Icons.sms_outlined),
                  title: Text('短信通知'),
                ),
              ],
            ),
            actions: [
              TDialogAction(
                child: Text('完成'),
                role: TDialogActionRole.primary,
              ),
            ],
          ),
        );
      },
    );
  }
