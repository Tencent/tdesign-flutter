
Widget _buildControl(BuildContext context) {
  var controller = TCountDownController();
  return Wrap(
    direction: Axis.vertical,
    spacing: 8,
    children: [
      Wrap(
        spacing: 8,
        children: [
          TButton(
            text: '开始',
            size: TButtonSize.extraSmall,
            theme: TButtonTheme.primary,
            onTap: () {
              controller.start();
            },
          ),
          TButton(
            text: '结束',
            size: TButtonSize.extraSmall,
            theme: TButtonTheme.primary,
            onTap: () {
              controller.reset(0);
            },
          ),
          TButton(
            text: '重置',
            size: TButtonSize.extraSmall,
            theme: TButtonTheme.primary,
            onTap: () {
              controller.reset();
            },
          ),
          TButton(
            text: '暂停',
            size: TButtonSize.extraSmall,
            theme: TButtonTheme.primary,
            onTap: () {
              controller.pause();
            },
          ),
          TButton(
            text: '继续',
            size: TButtonSize.extraSmall,
            theme: TButtonTheme.primary,
            onTap: () {
              controller.resume();
            },
          ),
        ],
      ),
      TCountDown(
        time: 60 * 60 * 1000,
        controller: controller,
        autoStart: false,
      ),
    ],
  );
}
