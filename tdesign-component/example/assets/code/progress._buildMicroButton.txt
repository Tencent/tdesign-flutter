
  Widget _buildMicroButton(BuildContext context) {
    return TProgress(
      type: TProgressType.micro,
      value: microProgressValue,
      onTap: _toggleMicroProgress,
      label: TIconLabel(
        isPlaying ? Icons.pause : Icons.play_arrow,
        color: TTheme.of(context).brandNormalColor,
      ),
    );
  }
