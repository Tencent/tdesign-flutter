Widget _buildHalfRate(BuildContext context) {
    return _rateCell(
      title: '半星评分',
      rate: Row(
        mainAxisSize: MainAxisSize.min,
        children: [
          TRate(
            value: _halfValue,
            allowHalf: true,
            onChanged: (value) => setState(() => _halfValue = value),
          ),
          const SizedBox(width: 8),
          TText(_halfValue.toStringAsFixed(1)),
        ],
      ),
    );
  }
