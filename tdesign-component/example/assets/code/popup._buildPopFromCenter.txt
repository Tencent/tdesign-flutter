
  Widget _buildPopFromCenter(BuildContext context) {
    return TButton(
      text: 'center',
      isBlock: true,
      theme: TButtonTheme.primary,
      type: TButtonType.outline,
      size: TButtonSize.large,
      onTap: () {
        TPopup.show(
          context,
          options: TPopupOptions.center(
              width: 240,
              height: 240,
              child: Container(
                color: TTheme.of(context).bgColorContainer,
              )),
        );
      },
    );
  }
