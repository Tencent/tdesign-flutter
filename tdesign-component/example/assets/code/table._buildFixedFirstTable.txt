Widget _buildFixedFirstTable(BuildContext context) {
    return TTable(
      key: const ValueKey('table-fixed-first'),
      columns: [
        _column(
          'title1',
          (row) => row.title1,
          width: 94,
          fixed: TTableColumnFixed.left,
        ),
        for (var index = 2; index <= 6; index++)
          _column('title$index', (_) => '内容', width: 120),
      ],
      data: _rows,
    );
  }
