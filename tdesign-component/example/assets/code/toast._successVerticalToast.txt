
  Widget _successVerticalToast(BuildContext context) {
    return TButton(
      onTap: () {
        TToast.showSuccess(
          '成功文案',
          direction: IconTextDirection.vertical,
          context: context,
        );
      },
      size: TButtonSize.large,
      type: TButtonType.outline,
      theme: TButtonTheme.primary,
      isBlock: true,
      text: '成功提示(竖向)',
    );
  }
